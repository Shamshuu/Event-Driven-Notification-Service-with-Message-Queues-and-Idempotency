docker compose up -d --build

docker compose ps

# 1. Health Check
curl -i http://localhost:8080/health

# 2. Successful notification (Email)
curl -X POST -H "Content-Type: application/json" -d '{
  "event_id": "11111111-2222-4333-8444-555555555555",
  "type": "email",
  "recipient": "user@example.com",
  "payload": {
    "subject": "Greetings!",
    "body": "This is a successful notification."
  },
  "timestamp": "'$(date -u +"%Y-%m-%dT%H:%M:%SZ")'"
}' http://localhost:8080/api/v1/publish-notification-event

# 3. Idempotency check (Duplicate event ID)
curl -X POST -H "Content-Type: application/json" -d '{
  "event_id": "11111111-2222-4333-8444-555555555555",
  "type": "email",
  "recipient": "user@example.com",
  "payload": {
    "subject": "Greetings!",
    "body": "This is a successful notification."
  },
  "timestamp": "'$(date -u +"%Y-%m-%dT%H:%M:%SZ")'"
}' http://localhost:8080/api/v1/publish-notification-event

# 4. Transient failure & retries leading to DLQ (SMS)
curl -X POST -H "Content-Type: application/json" -d '{
  "event_id": "22222222-3333-4444-9555-666666666666",
  "type": "sms",
  "recipient": "+15550199",
  "payload": {
    "body": "This event will fail transiently.",
    "simulate_transient_failure": true
  },
  "timestamp": "'$(date -u +"%Y-%m-%dT%H:%M:%SZ")'"
}' http://localhost:8080/api/v1/publish-notification-event

# 5. Permanent failure & direct DLQ routing (Push)
curl -X POST -H "Content-Type: application/json" -d '{
  "event_id": "33333333-4444-4555-a666-777777777777",
  "type": "push",
  "recipient": "invalid-device-token",
  "payload": {
    "title": "Immediate DLQ",
    "body": "This event will fail permanently.",
    "simulate_permanent_failure": true
  },
  "timestamp": "'$(date -u +"%Y-%m-%dT%H:%M:%SZ")'"
}' http://localhost:8080/api/v1/publish-notification-event

# 6. Schema validation failure (invalid type/UUID)
curl -i -X POST -H "Content-Type: application/json" -d '{
  "event_id": "invalid-uuid-string",
  "type": "slack",
  "recipient": "#general",
  "payload": {
    "body": "This will fail schema validation."
  },
  "timestamp": "'$(date -u +"%Y-%m-%dT%H:%M:%SZ")'"
